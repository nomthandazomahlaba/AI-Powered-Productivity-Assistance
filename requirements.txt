# 1. SMART EMAIL GENERATOR
# ==========================================
with tab1:
    st.header("Smart Email Generator")
    st.caption("Draft context-aware professional emails tailored by audience and tone.")
    
    col1, col2 = st.columns([1, 1], gap="large")
    
    with col1:
        audience = st.selectbox("Target Audience", ["Client / External Partner", "Executive / Manager", "Internal Team / Peers"])
        tone = st.selectbox("Desired Tone", ["Formal & Professional", "Informal & Conversational", "Persuasive & Value-Driven"])
        context = st.text_area("Context / Key Message Details", height=150, placeholder="e.g., Requesting a 2-week extension on project deadline due to delayed vendor API documentation...")
        
        generate_email = st.button("Generate Professional Draft", key="email_btn")
    
    with col2:
        st.subheader("Generated Email Draft")
        if generate_email:
            if context.strip():
                with st.spinner("Engineering prompt & generating draft..."):
                    time.sleep(0.8) # Simulate AI processing
                    
                    if "Formal" in tone and "Manager" in audience:
                        subject = f"Status Update & Action Request: {context[:35]}..."
                        body = f"Dear [Manager Name],\n\nI am writing to provide a formal status update regarding {context}.\n\nKey Highlights:\n- Progress is currently aligned with our primary Q3 targets.\n- Deliverables are undergoing final quality assurance checks.\n\nPlease let me know if you would like to adjust our current execution strategy.\n\nBest regards,\n[Your Name]"
                    elif "Persuasive" in tone:
                        subject = f"Unlocking Project Value: {context[:35]}..."
                        body = f"Hi [Client Name],\n\nI hope this message finds you well.\n\nI wanted to highlight an exciting opportunity regarding {context}. By aligning on these key deliverables today, we can streamline implementation and reduce operational overhead significantly.\n\nWould you be available for a brief 10-minute sync this Thursday?\n\nBest regards,\n[Your Name]"
                    else:
                        subject = f"Quick Sync: {context[:35]}..."
                        body = f"Hey team,\n\nQuick heads-up concerning {context}.\n\nLet's make sure we are all aligned on our priority targets for the week. Feel free to drop any blockers in the channel.\n\nCheers,\n[Your Name]"
                    
                    draft_text = f"SUBJECT: {subject}\n\n{body}"
                    st.code(draft_text, language="text")
                    
                    st.markdown("""
                        <div class="disclaimer-box">
                            ⚠️ <b>Human-in-the-Loop Review Required:</b> Please replace all bracketed placeholders <code>[ ]</code> with actual details before sending.
                        </div>
                    """, unsafe_allow_html=True)
            else:
                st.warning("Please provide context or details to generate an email.")
        else:
            st.info("Fill in the fields on the left and click 'Generate' to produce a draft.")

# 5. AI CHATBOT INTERFACE
# ==========================================
with tab5:
    st.header("Workplace AI Chatbot Assistant")
    st.caption("Interactive conversational assistant to handle quick workplace queries.")
    
    # Initialize chat history state
    if "messages" not in st.session_state:
        st.session_state.messages = [
            {"role": "assistant", "content": "Hello! I am your AI Workplace Assistant. How can I help you streamline your work today?"}
        ]

    # Display chat history
    for msg in st.session_state.messages:
        with st.chat_message(msg["role"]):
            st.markdown(msg["content"])

    # Handle user input
    if user_prompt := st.chat_input("Ask a question or request help with a task..."):
        # Display user message
        st.session_state.messages.append({"role": "user", "content": user_prompt})
        with st.chat_message("user"):
            st.markdown(user_prompt)

        # Generate assistant response
        with st.chat_message("assistant"):
            with st.spinner("Thinking..."):
                time.sleep(0.5)
                bot_response = f"I can certainly help with that! Regarding **'{user_prompt}'**, I recommend utilizing the specialized tabs above (such as the *Smart Email Generator* or *Task Planner*) for structured output, or let me know if you need a quick draft here!"
                st.markdown(bot_response)
                
        st.session_state.messages.append({"role": "assistant", "content": bot_response})
